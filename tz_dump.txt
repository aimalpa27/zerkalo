1. Общая архитектура

Поток данных (пример: гость делает заказ)
1. React отправляет POST /api/v1/sessions/
   с table_token и списком блюд

2. Django views.py получает запрос
   → валидирует через serializers.py
   → передаёт в services.py

3. services.py (бизнес-логика):
   → проверяет table_token в PostgreSQL
   → ищет открытую сессию стола
   → создаёт/обновляет TableSession
   → пересчитывает суммы

4. Django сохраняет в PostgreSQL

5. Django отправляет WS-событие через Redis pub/sub
   → все подключённые кухни и админы получают уведомление

6. Django возвращает JSON ответ фронтенду
2. Замена Firebase
2.1 Firebase Auth → Django Auth + JWT
Что было в Firebase:
signInWithEmailAndPassword(auth, email, password)
onAuthStateChanged(auth, callback)
createUserWithEmailAndPassword(secondaryAuth, email, pass)
Что станет в Django:
# Вход — POST /api/v1/auth/login/
# Django проверяет email+password, возвращает два токена:
{
  "access": "eyJhbGc...",   # живёт 60 минут
  "refresh": "eyJhbGc..."  # живёт 30 дней
}

# Каждый запрос к защищённым эндпоинтам:
Authorization: Bearer eyJhbGc...

# Создание сотрудника — POST /api/v1/staff/
# Администратор сам создаёт аккаунт, сам остаётся авторизован
# Никакого secondaryAuth не нужно — это была Firebase-специфика
Миграция пользователей:
Экспортировать из Firebase Auth через Firebase Admin SDK
Пароли Firebase хешированы через scrypt — нельзя перенести напрямую
Решение: при первом входе после миграции → пользователь вводит пароль заново
Подробно описано в разделе 15
2.2 Firestore → PostgreSQL
Что было (Firestore коллекции):
restaurants/{id}
  categories/{id}
  menuItems/{id}
  tableSessions/{id}
  tables/{id}
  waiterCalls/{id}
  promos/{id}
users/{id}
Что станет (PostgreSQL таблицы):
restaurants
categories
menu_items
table_sessions
session_items      ← важно: отдельная таблица, не JSONB массив
tables
waiter_calls
promos
users
Главное отличие: в Firestore SessionItem хранился как массив внутри документа TableSession. В PostgreSQL это отдельная таблица session_items с FK на table_sessions. Это правильный реляционный подход.

# apps/users/models.py
import uuid
from django.contrib.auth.models import AbstractBaseUser, BaseUserManager
from django.db import models

class UserManager(BaseUserManager):
    def create_user(self, email, password=None, **extra_fields):
        if not email:
            raise ValueError('Email обязателен')
        email = self.normalize_email(email)
        user = self.model(email=email, **extra_fields)
        user.set_password(password)
        user.save(using=self._db)
        return user

    def create_superuser(self, email, password=None, **extra_fields):
        extra_fields.setdefault('role', 'admin')
        return self.create_user(email, password, **extra_fields)


class User(AbstractBaseUser):
    ROLES = [
        ('admin',   'Администратор'),
        ('manager', 'Менеджер'),
        ('waiter',  'Официант'),
        ('cashier', 'Кассир'),
    ]

    id             = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    email          = models.EmailField(unique=True)
    name           = models.CharField(max_length=255)
    role           = models.CharField(max_length=10, choices=ROLES, default='waiter')
    restaurant     = models.ForeignKey(
                         'restaurants.Restaurant',
                         on_delete=models.SET_NULL,
                         null=True, blank=True,
                         related_name='staff'
                     )
    assigned_tables = models.ManyToManyField(
                         'tables.Table',
                         blank=True,
                         related_name='assigned_waiters'
                      )
    is_active      = models.BooleanField(default=True)
    created_at     = models.DateTimeField(auto_now_add=True)

    USERNAME_FIELD  = 'email'
    REQUIRED_FIELDS = ['name']
    objects         = UserManager()

    class Meta:
        db_table = 'users'
        indexes  = [models.Index(fields=['restaurant', 'role'])]

    def __str__(self):
        return f'{self.name} ({self.email})'

    @property
    def is_admin_or_manager(self):
        return self.role in ('admin', 'manager')

    is_available        = models.BooleanField(default=True)    # стоп-лист
    is_visible          = models.BooleanField(default=True)    # скрыть из меню
    badge               = models.CharField(max_length=10, choices=BADGES, blank=True)
    discount_percent    = models.PositiveIntegerField(default=0)
    preparation_station = models.CharField(max_length=10, choices=STATIONS, default='kitchen')
    sort_order          = models.PositiveIntegerField(default=0)

    created_at          = models.DateTimeField(auto_now_add=True)
    updated_at          = models.DateTimeField(auto_now=True)

    class Meta:
        db_table = 'menu_items'
        ordering = ['sort_order']
        indexes  = [
            models.Index(fields=['restaurant', 'is_available', 'is_visible']),
            models.Index(fields=['category']),
        ]

    @property
    def final_price(self):
        """Цена с учётом скидки"""
        if self.discount_percent > 0:
            return self.price - (self.price * self.discount_percent // 100)
        return self.price

4.6 TableSession
class TableSession(models.Model):
    STATUSES = [
        ('open',               'Открыт'),
        ('awaiting_payment',   'Ожидает оплату'),
        ('payment_requested',  'Запрошена оплата'),
        ('closed',             'Закрыт'),
    ]
    PAYMENT_METHODS = [('cash', 'Наличные'), ('card', 'Карта')]

    id                     = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    restaurant             = models.ForeignKey(Restaurant, on_delete=models.CASCADE, related_name='sessions')
    table                  = models.ForeignKey(Table, on_delete=models.SET_NULL, null=True, related_name='sessions')
    table_number           = models.PositiveIntegerField()          # денормализовано — стол могут удалить
    table_token            = models.CharField(max_length=64)        # денормализовано
    status                 = models.CharField(max_length=20, choices=STATUSES, default='open')
    subtotal_amount        = models.DecimalField(max_digits=10, decimal_places=2, default=0)
    service_charge_percent = models.PositiveIntegerField(default=10)
    service_charge_amount  = models.DecimalField(max_digits=10, decimal_places=2, default=0)
    total_amount           = models.DecimalField(max_digits=10, decimal_places=2, default=0)
    payment_method         = models.CharField(max_length=10, choices=PAYMENT_METHODS, blank=True)
    created_at             = models.DateTimeField(auto_now_add=True)
    updated_at             = models.DateTimeField(auto_now=True)
    closed_at              = models.DateTimeField(null=True, blank=True)

    class Meta:
        db_table = 'table_sessions'
        indexes  = [
            models.Index(fields=['restaurant', 'status']),       # поиск открытых сессий
            models.Index(fields=['table_token', 'status']),      # гость ищет свою сессию
            models.Index(fields=['restaurant', 'closed_at']),    # аналитика
        ]

    def recalculate_totals(self):
        """Пересчитать суммы на основе SessionItem"""
        items = self.items.exclude(status='cancelled')
        subtotal = sum(item.price * item.quantity for item in items)
        svc = subtotal * self.service_charge_percent // 100
        self.subtotal_amount = subtotal
        self.service_charge_amount = svc
        self.total_amount = subtotal + svc
        self.save(update_fields=['subtotal_amount', 'service_charge_amount', 'total_amount', 'updated_at'])

# apps/menu/views.py
from django.core.cache import cache

class MenuItemListView(generics.ListAPIView):
    serializer_class = MenuItemSerializer

    def get_queryset(self):
        rest_id = self.kwargs['rest_id']
        cache_key = f"menu:{rest_id}"

        cached = cache.get(cache_key)
        if cached is not None:
            return cached

        queryset = MenuItem.objects.filter(
            restaurant_id=rest_id,
            is_available=True,
            is_visible=True
        ).select_related('category').order_by('sort_order')

        cache.set(cache_key, queryset, timeout=300)  # 5 минут
        return queryset

# apps/menu/views.py
import cloudinary.uploader

class MenuItemImageUploadView(APIView):
    permission_classes = [IsAuthenticated, IsAdminOrManager]

    def post(self, request, rest_id, item_id):
        item = get_object_or_404(MenuItem, id=item_id, restaurant_id=rest_id)

        if 'image' not in request.FILES:
            return Response({'error': 'Файл не передан'}, status=400)

        file = request.FILES['image']

        # Проверка типа файла
        if not file.content_type.startswith('image/'):
            return Response({'error': 'Только изображения'}, status=400)

        # Проверка размера (максимум 10MB)
        if file.size > 10 * 1024 * 1024:
            return Response({'error': 'Файл слишком большой (максимум 10MB)'}, status=400)

        # Загрузка в Cloudinary
        result = cloudinary.uploader.upload(
            file,
            folder=f"plait/{rest_id}/menu",
            public_id=str(item_id),         # перезапишет старое фото
            overwrite=True,
            transformation=[
                {'width': 800, 'height': 600, 'crop': 'fill'},
                {'quality': 'auto', 'fetch_format': 'auto'}
            ]
        )

        # Сохраняем URL в БД
        item.image_url = result['secure_url']
        item.save(update_fields=['image_url', 'updated_at'])

        # Запускаем Celery задачу для дополнительной оптимизации
        from .tasks import optimize_menu_image
        optimize_menu_image.delay(str(item.id))

        return Response({'image_url': item.image_url})

Схема JWT
ВХОД:
POST /auth/login/ → { access (60 мин), refresh (30 дней) }
                           ↓
Фронтенд сохраняет токены в localStorage

ОБЫЧНЫЙ ЗАПРОС:
Authorization: Bearer {access}
                    ↓
Django декодирует токен → находит пользователя → выполняет запрос

ТОКЕН ИСТЁК:
401 Unauthorized
    ↓
Фронтенд автоматически: POST /auth/refresh/ с refresh токеном
    ↓
Получает новый access токен и повторяет запрос

ВЫХОД:
Фронтенд удаляет токены из localStorage
(опционально: POST /auth/logout/ для blacklist refresh токена)

from rest_framework_simplejwt.serializers import TokenObtainPairSerializer

class CustomTokenObtainPairSerializer(TokenObtainPairSerializer):
    @classmethod
    def get_token(cls, user):
        token = super().get_token(user)
        # Добавляем данные в payload — фронтенд их видит без запроса к /me/
        token['name'] = user.name
        token['role'] = user.role
        token['restaurant_id'] = str(user.restaurant_id) if user.restaurant_id else None
        return token

11.2 Защита от SQL injection
Django ORM автоматически экранирует все параметры. Никогда не использовать raw() с пользовательскими данными без параметризации:
# ❌ ОПАСНО:
User.objects.raw(f"SELECT * FROM users WHERE email = '{email}'")

# ✅ БЕЗОПАСНО — ORM сам экранирует:
User.objects.filter(email=email)

# ✅ БЕЗОПАСНО — если raw нужен, только параметризованный:
User.objects.raw("SELECT * FROM users WHERE email = %s", [email])

import json
from channels.generic.websocket import AsyncWebsocketConsumer
from rest_framework_simplejwt.tokens import AccessToken
from django.contrib.auth import get_user_model

User = get_user_model()

class RestaurantConsumer(AsyncWebsocketConsumer):
    async def connect(self):
        self.rest_id = self.scope['url_route']['kwargs']['rest_id']
        self.group_name = f"restaurant_{self.rest_id}"

        # Аутентификация через токен в query string
        token = self.scope['query_string'].decode().split('token=')[-1]
        try:
            payload = AccessToken(token)
            user = await User.objects.aget(id=payload['user_id'])
            if str(user.restaurant_id) != self.rest_id:
                await self.close(code=4003)   # Нет доступа к этому ресторану
                return
        except Exception:
            await self.close(code=4001)        # Невалидный токен
            return

        await self.channel_layer.group_add(self.group_name, self.channel_name)
        await self.accept()

    async def disconnect(self, close_code):
        await self.channel_layer.group_discard(self.group_name, self.channel_name)

    async def receive(self, text_data):
        """Обработка сообщений от клиента (кухня меняет статус через WS)"""
        data = json.loads(text_data)

        if data['type'] == 'mark_item_ready':
            await self.handle_mark_ready(data['session_id'], data['item_id'])

    # Получение событий от других частей Django и пересылка клиенту
    async def sessions_update(self, event):
        await self.send(text_data=json.dumps(event))

    async def new_order(self, event):
        await self.send(text_data=json.dumps(event))

    async def waiter_call(self, event):
        await self.send(text_data=json.dumps(event))

# migration/migrate_from_firebase.py

import firebase_admin
from firebase_admin import credentials, firestore
import django
import os

fs = firestore.client()

from apps.restaurants.models import Restaurant
from apps.menu.models import Category, MenuItem

def migrate_restaurants():
    docs = fs.collection('restaurants').stream()
    for doc in docs:
        data = doc.to_dict()
        rest, created = Restaurant.objects.update_or_create(
            slug=data.get('slug', doc.id),
            defaults={
                'name':                   data.get('name', ''),
                'address':                data.get('address', ''),
                'service_charge_percent': data.get('serviceChargePercent', 10),
                'is_public':              data.get('isPublic', True),
            }
        )
        print(f"{'Created' if created else 'Updated'}: {rest.name}")
        migrate_menu(doc.id, rest)

def migrate_menu(firebase_rest_id: str, rest: Restaurant):
    # Категории
    cat_map = {}   # firebase_id → Category
    cats = fs.collection(f'restaurants/{firebase_rest_id}/categories').stream()
    for doc in cats:
        d = doc.to_dict()
        cat = Category.objects.create(
            restaurant=rest,
            name=d.get('name', ''),
            sort_order=d.get('sortOrder', 0)
        )
        cat_map[doc.id] = cat

    # Блюда
    items = fs.collection(f'restaurants/{firebase_rest_id}/menuItems').stream()
    for doc in items:
        d = doc.to_dict()
        MenuItem.objects.create(
            restaurant=rest,
            category=cat_map.get(d.get('categoryId')),
            name=d.get('name', ''),
            description=d.get('description', ''),
            price=d.get('price', 0),
            image_url=d.get('imageUrl', ''),
            emoji=d.get('emoji', ''),
            weight=d.get('weight', ''),
            is_available=d.get('isAvailable', True),
            is_visible=d.get('isVisible', True),
            badge=d.get('badge') or '',
            discount_percent=d.get('discountPercent', 0),
            preparation_station=d.get('preparationStation', 'kitchen'),
            sort_order=d.get('sortOrder', 0),
        )

# migration/migrate_users.py
# Использует Firebase Admin SDK для получения списка пользователей

from firebase_admin import auth as firebase_auth
from apps.users.models import User
from apps.restaurants.models import Restaurant

def migrate_users():
    # Получить всех пользователей из Firestore (профили)
    user_docs = fs.collection('users').stream()

    for doc in user_docs:
        d = doc.to_dict()
        try:
            rest = Restaurant.objects.get(id=d.get('restaurantId'))
        except Restaurant.DoesNotExist:
            rest = None

        user, created = User.objects.get_or_create(
            email=d.get('email', ''),
            defaults={
                'name':       d.get('name', ''),
                'role':       d.get('role', 'waiter'),
                'restaurant': rest,
                'is_active':  True,
            }
        )

        if created:
            # Установить временный пароль, попросить сменить
            user.set_password('ChangeMe123!')
            user.save()
            print(f"Создан: {user.email} — пароль: ChangeMe123!")
